package shipeditor.communication;

import lombok.extern.log4j.Log4j2;
import shipeditor.communication.events.viewer.layers.LayerEvents.LayerWasSelected;
import shipeditor.components.viewer.layers.ViewerLayer;

@Log4j2
public class EventSubscriptionExample {

    private final Object parentComponent;

    public EventSubscriptionExample(Object parentComponent) {
        this.parentComponent = parentComponent;
    }

    public void registerListeners() {
        // Typed lifecycle-bound subscription. Automatically cleaned up when the parent is GC'd.
        EventBus.subscribe(parentComponent, LayerWasSelected.class, event -> {
            handleLayerSelected(event.selected());
        });
    }

    public void unregisterListeners() {
        EventBus.unsubscribeByParent(parentComponent);
    }

    private void handleLayerSelected(ViewerLayer selected) {
        if (selected == null) {
            return;
        }
        log.debug("Selected layer: {}", selected.getSprite());
    }
}
