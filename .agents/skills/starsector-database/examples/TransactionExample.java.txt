package shipeditor.persistence.database;

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.SQLException;
import java.util.List;

public class TransactionExample {

    public static void batchInsertFiles(Connection conn, List<IndexedFile> files) throws SQLException {
        String sql = "INSERT OR REPLACE INTO indexed_files " +
                     "(uuid, mod_id, entity_id, entity_name, entity_type, file_name, file_path, last_modified, file_hash, sprite_path, designation, metadata_json) " +
                     "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)";

        boolean autoCommit = conn.getAutoCommit();
        try {
            // Disable auto-commit for single-transaction performance
            conn.setAutoCommit(false);

            try (PreparedStatement pstmt = conn.prepareStatement(sql)) {
                int count = 0;
                for (IndexedFile file : files) {
                    pstmt.setString(1, file.getUuid().toString());
                    pstmt.setString(2, file.getModId());
                    pstmt.setString(3, file.getEntityId());
                    pstmt.setString(4, file.getEntityName());
                    pstmt.setString(5, file.getEntityType());
                    pstmt.setString(6, file.getFileName());
                    pstmt.setString(7, file.getFilePath().toString());
                    pstmt.setLong(8, file.getLastModified());
                    pstmt.setString(9, file.getFileHash());
                    pstmt.setString(10, file.getSpritePath());
                    pstmt.setString(11, file.getDesignation());
                    pstmt.setString(12, file.getMetadataJson());

                    pstmt.addBatch();
                    count++;

                    // Flush batch every 500 statements to balance memory and performance
                    if (count % 500 == 0) {
                        pstmt.executeBatch();
                    }
                }
                pstmt.executeBatch(); // final flush
            }

            conn.commit();
        } catch (SQLException e) {
            conn.rollback();
            throw e;
        } finally {
            conn.setAutoCommit(autoCommit);
        }
    }
}
